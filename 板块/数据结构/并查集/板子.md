# 并查集

## 基本并查集

### 代码实现
```cpp
class DSU {
    public: 
        vector <int> fa, rk, sz;
        DSU (int n = 0) {
            init(n); 
        }

        void init (int n) {
            fa.resize(n + 1);   // 祖先
            rk.assign(n + 1, 1);   // rk, 秩, 即深度
            sz.assign(n + 1, 1);   // 如果有多个树的话, 记录每个的节点数
            ranges::iota(fa, 0LL);  //将fa按照0, 1, 2, 3...赋值
        }

        int find (int x) {
            return x == fa[x] ? x : (fa[x] = find(fa[x])); // 寻找父节点, 如果x不是根节点则一直递归到fa[x]为x树的根节点
        }

        // 按秩合并
        bool merge (int i, int j) {
            int x = find(i), y = find(j);
            if (x == y) return false;
            if (rk[x] < rk[y]) swap(x, y);
            fa[y] = x;
            sz[x] += sz[y]; // 合并后sz增加
            if (rk[x] == rk[y]) rk[x]++;
            return true;
        }

        int size (int x) {
            return sz[find(x)];
        }
};
```

### 置换环应用

[基本置换环](https://ac.nowcoder.com/acm/contest/127702/F)

#### 思路
- 每次将i位置目前在的元素和需要在的元素合并(merge), 最终形成k颗树, 作为一个置换环

- 还原一个置换环, 步骤siz - 1(siz即置换环元素个数)

- 还原所有的, 即$\sum siz - 1$ = n - cnt (元素总个数 - 环个数)

#### 代码

```cpp
void ovo(){   
    int n; cin >> n;
    vector <int> a (n + 1);
    DSU dsu (n);
    
    for (int i = 1; i <= n; i ++) cin >> a[i];

    for (int i = 1; i <= n; i ++) {
        dsu.merge (i, a[i]);
    }

    int cnt = 0;
    for (int i = 1; i <= n; i ++) {
        if (dsu.find(i) == i) cnt ++;
    }

    int ans = n - cnt, ok = 0;
    for (int i = 1; i < n; i ++) {
        if (dsu.find(i) == dsu.find(i + 1)) {
            ok = 1; break;
        }
    }

    // 如果有相邻元素在一个置换环内, 则保持原样, 不在则需要交换一次
    cout << (ok ? (ans - 1) : (ans + 1)) << '\n';
}
```
