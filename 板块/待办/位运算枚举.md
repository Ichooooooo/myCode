# 位运算枚举

## 常见用法
```cpp
#include <bits/stdc++.h>
using namespace std;

/*
  位运算枚举常用板子
  约定：元素个数 n <= 60(用 unsigned long long)，若 n<=20/25 则 1<<n 枚举更常见。
*/

// 1) 枚举 mask 的所有子集 sub（含 0 和 mask）
// 用法：for (sub = mask; ; sub = (sub-1)&mask)
void enum_subsets(unsigned long long mask) {
    for (unsigned long long sub = mask;; sub = (sub - 1) & mask) {
        // use sub
        if (sub == 0) break;
    }
}

// 2) 枚举固定 popcount = k 的所有 mask（Gosper's Hack）
// 要求：n < 64，k <= n
// 生成的 mask 范围在 [0, 2^n)
void enum_k_bits(int n, int k) {
    if (k < 0 || k > n) return;
    if (k == 0) {
        unsigned long long mask = 0;
        // use mask
        return;
    }
    if (n >= 64) return; // 64-bit 装不下

    unsigned long long limit = (n == 64 ? ~0ULL : (1ULL << n));
    unsigned long long mask = (1ULL << k) - 1; // 最小的 k 个 1：000..0111..1

    while (mask < limit) {
        // use mask

        // next combination
        unsigned long long c = mask & -mask;
        unsigned long long r = mask + c;
        mask = (((r ^ mask) >> 2) / c) | r;
    }
}

// 3) 枚举所有 mask：0..(1<<n)-1（n <= 25 左右才这么干）
// 并遍历 mask 中所有为 1 的位（lowbit）
void enum_all_masks_and_bits(int n) {
    int N = 1 << n;
    for (int mask = 0; mask < N; mask++) {
        // 遍历 mask 的 set bits：O(popcount(mask))
        int t = mask;
        while (t) {
            int lb = t & -t;
            int bit = __builtin_ctz(lb); // 0-based
            // use bit
            t -= lb;
        }
    }
}

// 4) 一些常用小工具
int popcount_ull(unsigned long long x) {
    return __builtin_popcountll(x);
}
int lowbit_index_ull(unsigned long long x) { // x != 0
    return __builtin_ctzll(x); // 0-based
}

int main() {
    // 示例：n=5, k=3 枚举所有 3 个 1 的 mask
    enum_k_bits(5, 3);
    return 0;
}

```
