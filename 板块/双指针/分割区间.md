# 分割区间

## 常见写法

```cpp
int j = 0; 
for (int i = 1; i <= n + 1; ++i) {
    if (w[i]) { 
    v.push_back({j, i}); j = i; 
    }
}


vector<pair<int,int>> seg; // [l,r] 1-indexed
int l = 1;
for (int i = 1; i <= n; ++i) {
    bool cur = (a[i] != 0);
    bool pre = (a[i-1] != 0); // 需要 a[0]=0 或单独处理
    if (i == 1) { l = 1; continue; }
    if (cur != pre) {
        seg.push_back({l, i-1});
        l = i;
    }
}
seg.push_back({l, n});
```
