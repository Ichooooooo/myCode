vector<int>add(vector<int>&a, const vector<int>&b) {
   vector<int>res;
   int carry = 0;
   int w = max(a.size(), b.size());
   for (int i = 0; i < w; i++) {
     int sum = carry;
     if (i < a.size()) sum += a[i];
     if (i < b.size()) sum += b[i];
     res.push_back(sum % 10);
     carry = sum / 10;
   }
   if (carry > 0)
     res.push_back(carry);
   return res;
 }

 vector<int>multiply(vector<int>&a, const vector<int>&b) {
   int lenA = a.size();
   int lenB = b.size();
   vector<int>res(lenA + lenB, 0);

   for (int i = 0; i < lenA; i++) {
     for (int j = 0; j < lenB; j++) {
       res[i + j] += a[i] * b[j];
     }
   }
   int carry = 0;
   for (int k = 0; k < res.size(); k++) {
     int total = carry + res[k];
     res[k] = total % 10;
     carry = total / 10;
   }
   while(carry>0){
     res.push_back(carry%10);
     carry/=10;
   }
   while (res.size() > 1 && res.back() == 0) {
     res.pop_back();
   }
   return res;
 }

## stl补全

```cpp
vector\<T\>

动态数组，支持随机访问。

常用：push_back(x), pop_back(), size(), clear(), sort(all(v)).

string

字符串，支持 substr, find, stoi, to_string.

set / multiset

有序集合，自动去重（multiset 可重复）。

常用：insert(x), erase(x), find(x), lower_bound(x).

map / multimap

有序字典，键值对。

常用：m[key], m.count(key), erase(key).

unordered_set / unordered_map

哈希集合/字典，平均 O(1)。

常用：insert, find, erase.

queue / stack / deque

队列：push, pop, front, back.

栈：push, pop, top.

双端队列：push_front, push_back, pop_front, pop_back.

priority_queue\<T\>

默认大根堆。

小根堆写法：priority_queue<int, vector<int>, greater<int>> pq;.
```
