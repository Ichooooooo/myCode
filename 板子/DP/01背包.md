#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n, m;
    cin >> n >> m;

    vector<int> dp(m + 1, 0);

    for (int i = 1; i <= n; i++) {
        int cost, val;
        cin >> cost >> val;

        for (int j = m; j >= cost; j--) {
            dp[j] = max(dp[j], dp[j - cost] + val);
        }
    }

    cout << dp[m] << "\n";
}
